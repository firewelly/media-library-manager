## 3. 代码规范

- 使用4个空格缩进
- 每行最多79个字符
- 使用空行分隔函数和类
- 在二元运算符前后加空格
- 在逗号后加空格

## 4. 模块开发指南

### 4.1 数据库管理模块

#### 4.1.1 模块概述

数据库管理模块负责数据库连接、查询执行、事务管理等基础功能，是整个系统的数据访问层。

#### 4.1.2 核心类和方法

```python
class DatabaseManager:
    """
    数据库管理类
    
    提供数据库连接、查询执行、事务管理等功能。
    """
    
    def __init__(self, db_path):
        """
        初始化数据库管理器
        
        Args:
            db_path (str): 数据库文件路径
        """
        self.db_path = db_path
        self.connection = None
        self.logger = logging.getLogger(__name__)
        
    def get_connection(self):
        """
        获取数据库连接
        
        Returns:
            sqlite3.Connection: 数据库连接对象
        """
        if self.connection is None:
            self.connection = sqlite3.connect(self.db_path)
            self.connection.row_factory = sqlite3.Row  # 返回字典形式的结果
        return self.connection
    
    def close_connection(self):
        """关闭数据库连接"""
        if self.connection:
            self.connection.close()
            self.connection = None
    
    def execute_query(self, query, params=None):
        """
        执行查询
        
        Args:
            query (str): SQL查询语句
            params (list, optional): 查询参数
            
        Returns:
            list: 查询结果列表
        """
        try:
            conn = self.get_connection()
            cursor = conn.cursor()
            
            if params:
                cursor.execute(query, params)
            else:
                cursor.execute(query)
                
            # 将结果转换为字典列表
            results = [dict(row) for row in cursor.fetchall()]
            return results
            
        except sqlite3.Error as e:
            self.logger.error(f"执行查询失败: {str(e)}")
            self.logger.error(f"查询语句: {query}")
            if params:
                self.logger.error(f"查询参数: {params}")
            raise
    
    def execute_update(self, query, params=None):
        """
        执行更新
        
        Args:
            query (str): SQL更新语句
            params (list, optional): 更新参数
            
        Returns:
            int: 受影响的行数
        """
        try:
            conn = self.get_connection()
            cursor = conn.cursor()
            
            if params:
                cursor.execute(query, params)
            else:
                cursor.execute(query)
                
            conn.commit()
            return cursor.rowcount
            
        except sqlite3.Error as e:
            self.logger.error(f"执行更新失败: {str(e)}")
            self.logger.error(f"更新语句: {query}")
            if params:
                self.logger.error(f"更新参数: {params}")
            conn.rollback()
            raise
    
    @contextmanager
    def transaction(self):
        """
        事务上下文管理器
        
        使用方法:
            with db_manager.transaction():
                # 执行多个数据库操作
                db_manager.execute_update("INSERT INTO table (field) VALUES (?)", ["value"])
                db_manager.execute_update("UPDATE table SET field = ? WHERE id = ?", ["new_value", 1])
        """
        conn = self.get_connection()
        cursor = conn.cursor()
        
        try:
            cursor.execute("BEGIN TRANSACTION")
            yield
            conn.commit()
        except Exception as e:
            conn.rollback()
            self.logger.error(f"事务执行失败: {str(e)}")
            raise
```

#### 4.1.3 开发指南

1. **使用参数化查询**：始终使用参数化查询，避免SQL注入攻击。
2. **使用事务**：当需要执行多个相关操作时，使用事务确保数据一致性。
3. **错误处理**：捕获并记录数据库操作中的错误，提供有意义的错误信息。
4. **资源管理**：确保数据库连接在使用后正确关闭，可以使用上下文管理器。

```python
class VideoManager:
    """
    视频管理类
    
    负责视频信息的增删改查、视频分析、封面生成等操作。
    """
    
    def __init__(self, db_manager):
        """
        初始化视频管理器
        
        Args:
            db_manager (DatabaseManager): 数据库管理器实例
        """
        self.db_manager = db_manager
        self.logger = logging.getLogger(__name__)
    
    def add_video_file(self, file_path, title=None, description=None, rating=None):
        """
        添加视频文件
        
        Args:
            file_path (str): 视频文件路径
            title (str, optional): 视频标题
            description (str, optional): 视频描述
            rating (int, optional): 视频评分
            
        Returns:
            int: 新增视频的ID
            None: 添加失败
        """
        try:
            # 检查文件是否存在
            if not os.path.exists(file_path):
                self.logger.error(f"视频文件不存在: {file_path}")
                return None
                
            # 如果没有提供标题，使用文件名作为标题
            if not title:
                title = os.path.splitext(os.path.basename(file_path))[0]
                
            # 插入视频记录
            query = """
                INSERT INTO videos (file_path, title, description, rating, add_time)
                VALUES (?, ?, ?, ?, ?)
            """
            params = [
                file_path,
                title,
                description,
                rating,
                datetime.now()
            ]
            
            video_id = self.db_manager.execute_update(query, params)
            
            if video_id > 0:
                self.logger.info(f"成功添加视频: {title} (ID: {video_id})")
                return video_id
            else:
                self.logger.error(f"添加视频失败: {title}")
                return None
                
        except Exception as e:
            self.logger.error(f"添加视频时发生错误: {str(e)}")
            return None
    
    def get_video_by_id(self, video_id):
        """
        根据ID获取视频信息
        
        Args:
            video_id (int): 视频ID
            
        Returns:
            dict: 视频信息字典
            None: 如果视频不存在
        """
        try:
            query = "SELECT * FROM videos WHERE id = ?"
            results = self.db_manager.execute_query(query, [video_id])
            
            if results:
                return results[0]
            else:
                return None
                
        except Exception as e:
            self.logger.error(f"获取视频信息失败 (ID: {video_id}): {str(e)}")
            return None
    
    def get_all_videos(self, sort_by="add_time", sort_order="desc"):
        """
        获取所有视频
        
        Args:
            sort_by (str, optional): 排序字段
            sort_order (str, optional): 排序方式 (asc/desc)
            
        Returns:
            list: 视频信息列表
        """
        try:
            # 验证排序字段
            valid_sort_fields = ["id", "title", "file_path", "rating", "add_time", "update_time"]
            if sort_by not in valid_sort_fields:
                sort_by = "add_time"
                
            # 验证排序方式
            if sort_order.lower() not in ["asc", "desc"]:
                sort_order = "desc"
                
            query = f"SELECT * FROM videos ORDER BY {sort_by} {sort_order.upper()}"
            results = self.db_manager.execute_query(query)
            
            return results
            
        except Exception as e:
            self.logger.error(f"获取所有视频失败: {str(e)}")
            return []
    
    def get_videos_by_condition(self, filters=None, sort_by="add_time", sort_order="desc", limit=None, offset=None):
        """
        根据条件获取视频
        
        Args:
            filters (dict, optional): 过滤条件
            sort_by (str, optional): 排序字段
            sort_order (str, optional): 排序方式 (asc/desc)
            limit (int, optional): 限制返回数量
            offset (int, optional): 偏移量
            
        Returns:
            list: 视频信息列表
        """
        try:
            # 构建查询语句
            query = "SELECT * FROM videos"
            params = []
            
            # 添加过滤条件
            if filters:
                conditions = []
                
                for field, value in filters.items():
                    if isinstance(value, dict):
                        # 处理包含、大于、小于等条件
                        if "contains" in value:
                            conditions.append(f"{field} LIKE ?")
                            params.append(f"%{value['contains']}%")
                        elif "gt" in value:
                            conditions.append(f"{field} > ?")
                            params.append(value['gt'])
                        elif "gte" in value:
                            conditions.append(f"{field} >= ?")
                            params.append(value['gte'])
                        elif "lt" in value:
                            conditions.append(f"{field} < ?")
                            params.append(value['lt'])
                        elif "lte" in value:
                            conditions.append(f"{field} <= ?")
                            params.append(value['lte'])
                    else:
                        # 处理等于条件
                        conditions.append(f"{field} = ?")
                        params.append(value)
                
                if conditions:
                    query += " WHERE " + " AND ".join(conditions)
            
            # 验证排序字段
            valid_sort_fields = ["id", "title", "file_path", "rating", "add_time", "update_time"]
            if sort_by not in valid_sort_fields:
                sort_by = "add_time"
                
            # 验证排序方式
            if sort_order.lower() not in ["asc", "desc"]:
                sort_order = "desc"
                
            query += f" ORDER BY {sort_by} {sort_order.upper()}"
            
            # 添加分页
            if limit is not None:
                query += " LIMIT ?"
                params.append(limit)
                
                if offset is not None:
                    query += " OFFSET ?"
                    params.append(offset)
            
            results = self.db_manager.execute_query(query, params)
            
            return results
            
        except Exception as e:
            self.logger.error(f"根据条件获取视频失败: {str(e)}")
            return []

            else:

        except Exception as e:
```

```python
        except Exception as e:
            self.logger.error(f"添加标签到视频时发生错误: {str(e)}")
            return False
    
    def remove_tag_from_video(self, tag_id, video_id):
        """
        从视频中移除标签
        
        Args:
            tag_id (int): 标签ID
            video_id (int): 视频ID
            
        Returns:
            bool: 移除是否成功
        """
        try:
            # 检查关联是否存在
            existing = self.db_manager.execute_query(
                "SELECT * FROM video_tags WHERE tag_id = ? AND video_id = ?",
                [tag_id, video_id]
            )
            
            if not existing:
                self.logger.warning(f"标签与视频的关联不存在 (标签ID: {tag_id}, 视频ID: {video_id})")
                return False
            
            # 移除关联
            query = "DELETE FROM video_tags WHERE tag_id = ? AND video_id = ?"
            params = [tag_id, video_id]
            
            result = self.db_manager.execute_update(query, params)
            
            if result > 0:
                self.logger.info(f"成功从视频中移除标签 (标签ID: {tag_id}, 视频ID: {video_id})")
                return True
            else:
                self.logger.error(f"从视频中移除标签失败 (标签ID: {tag_id}, 视频ID: {video_id})")
                return False
                
        except Exception as e:
            self.logger.error(f"从视频中移除标签时发生错误: {str(e)}")
            return False
    
    def get_video_tags(self, video_id):
        """
        获取视频的标签
        
        Args:
            video_id (int): 视频ID
            
        Returns:
            list: 标签信息列表
        """
        try:
            query = """
                SELECT t.* FROM tags t
                JOIN video_tags vt ON t.id = vt.tag_id
                WHERE vt.video_id = ?
                ORDER BY t.name ASC
            """
            results = self.db_manager.execute_query(query, [video_id])
            
            return results
            
        except Exception as e:
            self.logger.error(f"获取视频标签失败 (视频ID: {video_id}): {str(e)}")
            return []
    
    def batch_add_tags_to_video(self, video_id, tag_names):
        """
        批量添加标签到视频
        
        Args:
            video_id (int): 视频ID
            tag_names (list): 标签名称列表
            
        Returns:
            dict: 结果字典，包含成功和失败的标签
        """
        try:
            results = {
                "success": [],
                "failed": []
            }
            
            for tag_name in tag_names:
                # 获取或创建标签
                tag = self.get_tag_by_name(tag_name)
                if not tag:
                    tag_id = self.create_tag(name=tag_name)
                    if not tag_id:
                        results["failed"].append(tag_name)
                        continue
                else:
                    tag_id = tag['id']
                
                # 添加标签到视频
                success = self.add_tag_to_video(tag_id=tag_id, video_id=video_id)
                if success:
                    results["success"].append(tag_name)
                else:
                    results["failed"].append(tag_name)
            
            return results
            
        except Exception as e:
            self.logger.error(f"批量添加标签到视频时发生错误: {str(e)}")
            return {"success": [], "failed": tag_names}
```

```python
import logging
import os

def setup_logging(log_level=logging.INFO, log_file=None):
    """
    设置日志配置
    
    Args:
        log_level (int): 日志级别
        log_file (str, optional): 日志文件路径
    """
    # 创建日志格式
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    )
    
    # 配置根日志记录器
    root_logger = logging.getLogger()
    root_logger.setLevel(log_level)
    
    # 清除现有的处理器
    for handler in root_logger.handlers[:]:
        root_logger.removeHandler(handler)
    
    # 添加控制台处理器
    console_handler = logging.StreamHandler()
    console_handler.setLevel(log_level)
    console_handler.setFormatter(formatter)
    root_logger.addHandler(console_handler)
    
    # 添加文件处理器
    if log_file:
        # 确保日志目录存在
        log_dir = os.path.dirname(log_file)
        if log_dir and not os.path.exists(log_dir):
            os.makedirs(log_dir)
        
        file_handler = logging.FileHandler(log_file)
        file_handler.setLevel(log_level)
        file_handler.setFormatter(formatter)
        root_logger.addHandler(file_handler)

# 使用示例
setup_logging(log_level=logging.DEBUG, log_file="logs/media_library.log")
```
